import { z } from "zod";
export const tradingRulesSchema=z.object({
 monthlyEnabled:z.boolean().default(false),
 smartMode:z.enum(["OFF","SUGGEST_ONLY","AUTO_APPLY"]).default("SUGGEST_ONLY"),
 competitiveBand:z.coerce.number().min(0).max(20).default(2),
 aboveMarketPercent:z.coerce.number().min(0).max(50).default(5),
 underpricedPercent:z.coerce.number().min(0).max(50).default(5),
 promotionOutlierPercent:z.coerce.number().min(1).max(80).default(15),
 minimumRetailers:z.coerce.number().int().min(3).max(10).default(3),
 maxAgeDays:z.coerce.number().int().min(1).max(60).default(35),
 maxProductsPerSession:z.coerce.number().int().min(1).max(20000).default(5000),
 maxRequestsPerSession:z.coerce.number().int().min(1).max(60000).default(6000),
 batchSize:z.coerce.number().int().min(1).max(50).default(10),
 concurrency:z.coerce.number().int().min(1).max(3).default(2),
 requestsPerMinute:z.coerce.number().int().min(1).max(60).default(10),
 maxAttempts:z.coerce.number().int().min(1).max(3).default(3),
 autoScopes:z.array(z.object({type:z.enum(["PRODUCT","BRAND","CATEGORY","PRICE_BAND"]),value:z.string().max(200),minimum:z.number().nonnegative().default(0),maximum:z.number().nonnegative().default(1000000)})).max(30).default([]),
}).refine(v=>v.smartMode!=="AUTO_APPLY"||v.autoScopes.length>0,"Auto Apply requires explicit scopes; global Auto Apply is prohibited.");
export type TradingRules=z.infer<typeof tradingRulesSchema>;
export const DEFAULT_TRADING_RULES=tradingRulesSchema.parse({});
